$secureKey = Read-Host "Enter your Groq API key" -AsSecureString

$ptr = [System.Runtime.InteropServices.Marshal]::SecureStringToBSTR($secureKey)

Write-Host "`nDone! Created: $dir\secrets.toml" -ForegroundColor Green
